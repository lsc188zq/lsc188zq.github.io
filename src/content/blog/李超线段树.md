#include<bits/stdc++.h>
using namespace std;

#define fi first
#define se second
#define mp make_pair
#define pb push_back
#define ll long long


template<typename T>
void read(T& x) {
    x = 0; bool f=0; char ch=getchar();
    while (ch<'0'||ch>'9') {
        if (ch=='-') f=1;
        ch=getchar();
    }
    while (ch>='0'&&ch<='9') x=x*10+ch-'0',ch = getchar();
    x=f?-x:x;
}

template<typename T>
void write(T x) {
    if (x < 0) putchar('-'), x = -x;
    if (x > 9) write(x / 10);
    putchar(x % 10 + '0');
}

template<typename T>
void write(T x, char c) {
    write(x);
    putchar(c);
}
const int MAXN=100010,MAXL=39989,MAXH=1e9;
const double eps=1e-10;

int n,ans,cnt,op,x;
int lx[MAXN],rx[MAXN],ly,ry;
double k[MAXN],b[MAXN];
double pos(int x,int y){
    return k[y]*x+b[y];
}
struct node{
    int l,r,k;
};
struct bst{
    node t[MAXL<<2];
    void build(int p,int l,int r){
        t[p].l=l; t[p].r=r;
        if (l==r) return;
        int mid=l+r>>1;
        build(p<<1,l,mid); build(p<<1|1,mid+1,r);
    }
    void insert(int p,int x){
        if (t[p].l>rx[x]||t[p].r<lx[x]) return;
        if (t[p].l>=lx[x]&&t[p].r<=rx[x]){
            if (!t[p].k) t[p].k=x;
            else{
                int a=t[p].k,b=x;
                int mid=t[p].l+t[p].r>>1;
                if (pos(mid,b)-pos(mid,a)>eps) swap(a,b);
                if (pos(t[p].l,b)-pos(t[p].l,a)>eps||((abs(pos(t[p].l,b)-pos(t[p].l,a)))<eps&&b<a)) insert(p<<1,b);
                else if (pos(t[p].r,b)-pos(t[p].r,a)>eps||((abs(pos(t[p].r,b)-pos(t[p].r,a)))<eps&&b<a)) insert(p<<1|1,b);
                t[p].k=a;
            }
            return;
        }
        insert(p<<1,x); insert(p<<1|1,x);
    }
    int qus(int p,int x){
        if (t[p].l==t[p].r) return t[p].k;
        int mid=t[p].l+t[p].r>>1;
        int y;
        if (x<=mid) y=qus(p<<1,x);
        else y=qus(p<<1|1,x);
        if (abs(pos(x,t[p].k)-pos(x,y))<eps) return min(t[p].k,y);
        return pos(x,t[p].k)-pos(x,y)>eps?t[p].k:y;
    }
}tree;
int main() {
    read(n);
    tree.build(1,1,MAXL);
    for (int i=1;i<=n;i++){
        read(op);
        if (op==0) read(x),x=(x+ans-1)%MAXL+1,ans=tree.qus(1,x),write(ans,'\n');
        else {
            ++cnt;
            read(lx[cnt]),read(ly),read(rx[cnt]),read(ry);
            lx[cnt]=(lx[cnt]+ans-1)%MAXL+1; ly=(ly+ans-1)%MAXH+1; 
            rx[cnt]=(rx[cnt]+ans-1)%MAXL+1; ry=(ry+ans-1)%MAXH+1; 
            if (lx[cnt]>rx[cnt]) swap(lx[cnt],rx[cnt]),swap(ly,ry);
            if (lx[cnt]==rx[cnt]) k[cnt]=0,b[cnt]=max(ly,ry);
            else{
                k[cnt]=(double)(ry-ly)/(rx[cnt]-lx[cnt]);
                b[cnt]=ly-k[cnt]*lx[cnt];
            }
            tree.insert(1,cnt);
        }
    }
    return 0;
}
