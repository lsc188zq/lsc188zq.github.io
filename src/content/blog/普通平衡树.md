## 前言
平衡树用的一般不多，但是还是必须要牢牢掌握的知识，在与其它数据结构结合时也能发挥出很大的作用（比如树套树）。
这一阶段我应该暂时只会用Treap来实现平衡树，或者直接使用std标准或者pbds库代替好了，暂时没有精力和时间去学习其它平衡树。

## 二叉查找树
### 简介
**二叉查找树** 是一种 **树形数据结构** ，支持插入，删除，查询等操作。
### 原理
对于每个节点，它左侧的子树的任意节点的值都小于该节点的值，右侧子树的任意节点的值都大于该节点的值。
这样，只要根据给定的值的大小，在二叉查找树上游走到对应的位置，就可以轻松实现上述操作了。
### 缺陷
对于一颗有 $n$ 个节点的二叉查找树在随机数据下，树高为 $\mathcal{O}(\log n)$ 级别，这样一次操作也是 $\mathcal{O}(\log n)$ 的时间复杂度。然而不难将树高卡到 $\mathcal{O}(n)$ 级别，单次操作时间复杂度退化为 $\mathcal{O}(n)$ 。比如插入序列单调递增这样。

## 平衡树
### 导入
为了解决二叉查找树的局限性，我们需要调整其树高为 $\mathcal{O}(\log n)$  。平衡树就是为了解决这一问题而诞生的。
### Treap简介
Treap 是一种 **弱平衡** 的 **二叉搜索树**。它基于 **二叉堆** 的思想，通过随机赋上第二权值，使得第二权值满足二叉堆的性质，以此来达到 **期望** 树高为 $\mathcal{O}(\log n)$ 。（~~应该也不会有人人品会差到被卡。~~）

## 旋转Treap
旋转Treap是通过 **左旋**  和 **右旋** 等旋转操作进行调整树的形态的使其满足二叉堆的性质，这两种操作不会违反二叉查找树的性质。
润一下oi-wiki上的图：
![旋转操作](https://oi-wiki.org/ds/images/treap-rotate.svg)
插入在叶子处进行，然后通过旋转进行调整。
删除则是将节点旋转到叶子，再直接删除。
具体可以看下面的代码。
旋转Treap的特点就是常数小，然而拓展性不高。

### 无旋Treap
无旋Treap的思想就是根据值或者排名然后分裂Treap进行一系列操作后再合并。根据分裂出来的树（我们要修改/查询的节点在叶子上）我们可以轻松地在上面进行各种操作。
感觉不难理解，可以看代码。
无旋Treap的特点就是常数大，但是拓展性高，比如可以可持久化，还可以实现区间操作之类。
## 代码

旋转Treap代码：（以前写的，可能没有那么好看）
```cpp
#include<bits/stdc++.h>
using namespace std;
const int INF=1e9,MAXN=100010;
mt19937 mr;//随机数生成器
int tot,n,root,m;//节点总数，询问数，根节点
struct node{
    int n;//值
    unsigned int x;//随机值
    int l;//左子节点
    int r;//右子节点
    int cnt;//该数数量
    int siz;//子树大小
}treap[MAXN];
inline int add(const int& val){//建立新节点
    treap[++tot].n=val;
    treap[tot].x=mr();
    treap[tot].siz=treap[tot].cnt=1;
    return tot;
}
inline void upd(int p){//更新子树大小
    if (p==0) return;
    treap[p].siz=treap[treap[p].l].siz+treap[treap[p].r].siz+treap[p].cnt;
}
inline void zig(int &p){//右旋
    int q=treap[p].l;
    treap[p].l=treap[q].r; treap[q].r=p; p=q;
    upd(treap[p].r); upd(p);
}
inline void zag(int &p){//左旋
    int q=treap[p].r;
    treap[p].r=treap[q].l; treap[q].l=p; p=q;
    upd(treap[p].l); upd(p);
}
inline void insert(int &p,const int& val){//插入新节点
    if (p==0) {p=add(val);upd(p);return;};
    if (val==treap[p].n) {treap[p].cnt++,upd(p);return;}
    if (val<treap[p].n){
        insert(treap[p].l,val);
        if (treap[treap[p].l].x>treap[p].x) zig(p);//大根堆随机化调整
    }
    else{
        insert(treap[p].r,val);
        if (treap[treap[p].r].x>treap[p].x) zag(p);
    }
    upd(p);
}
inline int search(int p,const int& val){
    if (!p) return 0;
    else if (treap[p].n==val) return p;
    else if (treap[p].n>val) return search(treap[p].l,val);
    else return search(treap[p].r,val);
}
inline void del(int &p,const int& val){
    if (p==0) return;
    if (val==treap[p].n){
        if (treap[p].cnt>1){treap[p].cnt--; upd(p); return;}
        if (treap[p].l||treap[p].r){
            if (treap[p].r==0||treap[treap[p].l].x>treap[treap[p].r].x) zig(p),del(treap[p].r,val);
            else zag(p),del(treap[p].l,val);
            upd(p);
        }
        else p=0;
        return;
    }
    if (val<treap[p].n) del(treap[p].l,val);
    else del(treap[p].r,val);
    upd(p);
}
inline int getprev(int p,const int& val){
    int ans=2;
    while (p){
        if (val==treap[p].n){
            if (treap[p].l){
                p=treap[p].l;
                while (treap[p].r>0) p=treap[p].r;
                ans=p;
            }
            break;
        }
        if (treap[p].n<val&&treap[p].n>treap[ans].n) ans=p;
        if (val<treap[p].n) p=treap[p].l;
        else p=treap[p].r;
    }
    return treap[ans].n;
}
inline int getnext(int p,const int& val){
    int ans=1;
    while (p){
        if (val==treap[p].n){
            if (treap[p].r){
                p=treap[p].r;
                while (treap[p].l>0) p=treap[p].l;
                ans=p;
            }
            break;
        }
        if (treap[p].n>val&&treap[p].n<treap[ans].n) ans=p;
        if (val<treap[p].n) p=treap[p].l;
        else p=treap[p].r;
    }
    return treap[ans].n;
}
inline int grade(int p,const int& val){
    if (!p) return 0;
    else if (treap[p].n==val) return treap[treap[p].l].siz;
    else if (treap[p].n>val) return grade(treap[p].l,val);
    else return grade(treap[p].r,val)+treap[treap[p].l].siz+treap[p].cnt;
}
inline int get(int p,const int& x){
    int nl=treap[treap[p].l].siz,cnt=treap[p].cnt;
    if (!p) return INF;
    else if (nl<x&&nl+cnt>=x) return treap[p].n;
    else if (nl>=x) return get(treap[p].l,x);
    else return get(treap[p].r,x-nl-cnt);
}
int main(){
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin>>n;
    mr.seed(time(0));//种子
    add(INF); root=1;
    insert(root,-INF);
    upd(root);
    while (n--){
        int op,x;
        cin>>op>>x;
        switch (op){
            case 1:
                insert(root,x);
                break;
            case 2:
                del(root,x);
                break;
            case 3:
                cout<<grade(root,x)<<"\n";
                break;
            case 4:
                cout<<get(root,x+1)<<"\n";
                break;
            case 5:
                cout<<getprev(root,x)<<"\n";
                break;
            case 6:
                cout<<getnext(root,x)<<"\n";
                break;
        }
    }
}
```

无旋Treap代码：
```cpp
#include<bits/stdc++.h>
using namespace std;

#define ull unsigned long long
#define fi first
#define se second
#define mp make_pair

const int MAXN=100010;

struct node{
    int l,r; //左右儿子
    int val,siz,cnt; //真值，子树大小，这个数个数
    ull pri; //大根堆优先度
};
struct treap{
    mt19937_64 rand; //随机数生成器
    int cnt,root;//节点数，根
    node t[MAXN];
    treap();
    int newnode(int val);//生成新节点
    void upd(int p);//更新子树大小
    pair<int,int> split(int p,int key);//分裂
    tuple<int,int,int> split_by_rk(int p,int rk);//根据排名分裂
    int merge(int u,int v);//合并节点
    void insert(int val);//插入节点
    void del(int val);//删除节点
    int qus_by_val(int& p,int val);//给值查询排名
    int qus_by_rk(int& p,int rk);//给排名查询值
    int qus_pre(int val);//查询前驱
    int qus_nxt(int val);//查询后驱
}tree;

template<typename T>
void read(T& x);
template<typename T>
void write(T x);
template<typename T>
void write(T x,char c);


int n,op,x;

int main(){
    read(n);
    for (int i=1;i<=n;i++){
        read(op); read(x);
        if (op==1) tree.insert(x);
        else if (op==2) tree.del(x);
        else if (op==3) write(tree.qus_by_val(tree.root,x),'\n');
        else if (op==4) write(tree.qus_by_rk(tree.root,x),'\n');
        else if (op==5) write(tree.qus_pre(x),'\n');
        else write(tree.qus_nxt(x),'\n');
    }
    return 0;
}

template<typename T>
void read(T& x){
    x=0; bool f=0; char ch=getchar();
    while (ch>'9'||ch<'0') {
        if (ch=='-') f=1;
        ch=getchar();
    }
    while (ch>='0'&&ch<='9') x=x*10+ch-'0',ch=getchar();
    x=f?-x:x;
}

template<typename T>
void write(T x){
    if (x<0) putchar('-'),x=-x;
    if (x>9) write(x/10);
    putchar(x%10+'0');
}

template<typename T>
void write(T x,char c){
    write(x); 
    putchar(c);
}

treap::treap(){
    rand.seed(time(0));
}

int treap::newnode(int val){ //生成新节点
    ++cnt; 
    t[cnt].val=val; 
    t[cnt].cnt=t[cnt].siz=1;
    t[cnt].pri=rand();
    return cnt;
}

void treap::upd(int p){ //更新子树大小
    t[p].siz=t[t[p].l].siz+t[t[p].r].siz+t[p].cnt;
}

pair<int,int> treap::split(int p,int key){ //分裂
    if (!p) return mp(0,0);
    if (t[p].val<=key){
        auto temp=split(t[p].r,key); //在右边，继续裂
        t[p].r=temp.fi;
        upd(p);
        return mp(p,temp.se);
    }
    else{
        auto temp=split(t[p].l,key); //在左边，继续裂
        t[p].l=temp.se;
        upd(p);
        return mp(temp.fi,p); 
    }
}

tuple<int,int,int> treap::split_by_rk(int p,int rk){ //根据排名分裂
    if (!p) return {0,0,0};
    int siz=t[t[p].l].siz;
    int l,r,mid;
    if (rk<=siz){ //在左边，继续裂
        tie(l,mid,r)=split_by_rk(t[p].l,rk);
        t[p].l=r;
        upd(p);
        return {l,mid,p};
    }
    else if (rk<=siz+t[p].cnt){ //在中间，返回
        l=t[p].l; r=t[p].r;
        t[p].l=0; t[p].r=0;
        upd(p);
        return {l,p,r};
    }
    else{ //在右边，继续裂，别忘了差分！！！
        tie(l,mid,r)=split_by_rk(t[p].r,rk-t[t[p].l].siz-t[p].cnt); 
        t[p].r=l;
        upd(p);
        return {p,mid,r};
    }
}

int treap::merge(int u,int v){ //合并节点
    if (!u) return v;
    if (!v) return u;
    if (t[u].pri>t[v].pri){ //满足大根堆性质，u为根
        t[u].r=merge(t[u].r,v);
        upd(u);
        return u;
    }
    else {
        t[v].l=merge(u,t[v].l);//v为根
        upd(v);
        return v;
    }
}

void treap::insert(int val){//插入节点
    auto temp1=split(root,val);
    auto temp2=split(temp1.fi,val-1); //先分裂
    if (!temp2.se) temp2.se=newnode(val);
    else t[temp2.se].cnt++,upd(temp2.se);
    int ft=merge(temp2.fi,temp2.se); 
    root=merge(ft,temp1.se);  //再合并
}

void treap::del(int val){ //删除节点
    auto temp1=split(root,val);
    auto temp2=split(temp1.fi,val-1); //先分裂
    t[temp2.se].cnt--; upd(temp2.se);
    if (!t[temp2.se].cnt) temp2.se=0;
    int ft=merge(temp2.fi,temp2.se); 
    root=merge(ft,temp1.se);  //再合并
}

int treap::qus_by_val(int& p,int val){ //给值查询排名
    auto temp=split(p,val-1);
    int ans=t[temp.fi].siz+1;
    p=merge(temp.fi,temp.se);
    return ans;
}

int treap::qus_by_rk(int& p,int rk){ //给排名查询值
    int l,r,mid;
    tie(l,mid,r)=split_by_rk(p,rk);
    int ans=t[mid].val;
    p=merge(merge(l,mid),r);
    return ans;
}

int treap::qus_pre(int val){ //查询前驱
    auto temp=split(root,val-1);
    int ans=qus_by_rk(temp.fi,t[temp.fi].siz);
    root=merge(temp.fi,temp.se);
    return ans;
}

int treap::qus_nxt(int val){ //查询后驱
    auto temp=split(root,val);
    int ans=qus_by_rk(temp.se,1);
    root=merge(temp.fi,temp.se);
    return ans;
}
```
