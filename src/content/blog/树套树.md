## 引入
对于维护二维信息问题，如果可离线的静态问题可以加一个CDQ分治或者扫描线之类的去掉一维，然后再用数据结构去处理另一维。如果是强制在线或者动态问题，要怎么办呢。考虑类似的思想，我们能否用先用一种数据结构去掉一维，内部再套另一种数据去掉另一维呢？

## 树套树
这时候就是树套树出场的时候了。树套树，顾名思义，就是一种树内部套了另一种树，种类很多，比如有线段树套平衡树，平衡树套平衡树，树状数组套平衡树，线段树套线段树等等。这样做很容易证明空间复杂度是 $n\log n$ 的（动态开点之类另算），至于时间复杂度，就要看具体操作的嵌套了。  
先上模板题。
### 【模板】树套树

#### 题目描述

您需要写一种数据结构（可参考题目标题），来维护一个有序数列，其中需要提供以下操作：

1. 查询 $k$ 在区间内的排名

2. 查询区间内排名为 $k$ 的值

3. 修改某一位置上的数值

4. 查询 $k$ 在区间内的前驱（前驱定义为严格小于 $x$，且最大的数，**若不存在输出 `-2147483647`**）

5. 查询 $k$ 在区间内的后继（后继定义为严格大于 $x$，且最小的数，**若不存在输出 `2147483647`**）

#### 输入格式

第一行两个数 $n,m$，表示长度为 $n$ 的有序序列和 $m$ 个操作。

第二行有 $n$ 个数，表示有序序列。

下面有 $m$ 行，$opt$ 表示操作标号。

若 $opt=1$，则为操作 $1$，之后有三个数 $l~r~k$，表示查询 $k$ 在区间 $[l,r]$ 的排名。

若 $opt=2$，则为操作 $2$，之后有三个数 $l~r~k$，表示查询区间 $[l,r]$ 内排名为  $k$ 的数。

若 $opt=3$，则为操作 $3$，之后有两个数 $pos~k$，表示将 $pos$ 位置的数修改为 $k$。

若 $opt=4$，则为操作 $4$，之后有三个数 $l~r~k$，表示查询区间 $[l,r]$ 内 $k$ 的前驱。

若 $opt=5$，则为操作 $5$，之后有三个数 $l~r~k$，表示查询区间 $[l,r]$ 内 $k$ 的后继。

#### 输出格式

对于操作 $1,2,4,5$，各输出一行，表示查询结果。

#### 提示

$1\le n,m\le5\times 10^4$，序列中的值在任何时刻 $\in[0,10^8]$。

#### 题解
我们可以根据位置来建出线段树，内部套平衡树维护该区间内所有节点。  
对于操作1，只需要对 $\mathcal{O}(\log n)$ 个被覆盖节点分别求出内部小于 $k$  的数然后相加，最后加 $1$ 即可，时间复杂度 $\mathcal{O}(\log^2 n)$ 。
至于操作 3，4，5，都能用类似方法解决。只有2有些麻烦，很明显可以二分后转化为1问题来解决，可那样复杂度是 $\mathcal{O}(\log^3 n)$ 的。  
事实上，我们可以换一种思路，外层数维护值域，内层树维护索引，可以用动态开点权值线段树套平衡树实现。改操作为扫一下值域内有多少数在给定区间内就可以，然后求排名在权值线段树内二分，单次操作复杂度均为 $\mathcal{O}(\log^2 n)$ 。
## 拓展
类似的思路，我们可以用这种方法再进行CDQ分治或者使用扫描线做到处理离线静态三维问题（尽量使用扫描线，不然 $\log$ 太多复杂度太高）。更高维的问题也能通过嵌套类似的方法解决。
## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;

#define ull unsigned long long
#define fi first
#define se second
#define mp make_pair
#define pb push_back

template<typename T>
void read(T& x);
template<typename T>
void write(T x);
template<typename T>
void write(T x,char c);

const int MAXN=50010,L=-1,R=1e8+1;
mt19937_64 ran(time(nullptr)); //随机数生成器
struct node_treap{
    int l,r; //左右儿子
    int pos; //位置
    int siz; //子树大小
    ull pri; //大根堆优先度
}tr[MAXN<<6];

struct treap{ //平衡树
    int root; //根
    treap();
    pair<int,int> split(int p,int pos);//分裂
    int newnode(int pos);//生成新节点
    int merge(int u,int v);//合并节点
    void upd(int p);//更新子树大小
    void insert(int pos);//插入节点
    void erase(int pos);//删除节点
    int qus(int l,int r); //查询该区间中的数个数
};

struct node_bst{
    int l,r;
    treap tr;
};
struct bst{ //动态开点权值线段树（内套平衡树）
    int cnt;
    node_bst t[MAXN<<6];
    bst();
    int newnode(); //新建节点
    int add(int p,int l,int r,int k,int v); //对应位置添加值
    int del(int p,int l,int r,int k,int v); //对应位置减少值
    int qus_num(int p,int l,int r,int ql,int qr,int pl,int pr); //询问区间数的个数
    int qus_rk(int p,int l,int r,int pl,int pr,int k); //查询区间排名为某值的数
    int qus_pre(int pl,int pr,int k); //查询区间数的前驱
    int qus_nxt(int pl,int pr,int k); //查询区间数的后驱
}tree;

int n,m,op,l,r,p,k,a[MAXN];
int cnt=0;

int main(){
    read(n); read(m);
    for (int i=1;i<=n;i++){
        read(a[i]); 
        tree.add(1,L,R,i,a[i]);
    }
    for (int i=1;i<=m;i++){
        read(op);
        if (op==1) read(l),read(r),read(k),write(tree.qus_num(1,L,R,L,k-1,l,r)+1,'\n');
        else if (op==2) read(l),read(r),read(k),write(tree.qus_rk(1,L,R,l,r,k),'\n');
        else if (op==3) read(p),read(k),tree.del(1,L,R,p,a[p]),a[p]=k,tree.add(1,L,R,p,a[p]);
        else if (op==4) read(l),read(r),read(k),write(tree.qus_pre(l,r,k),'\n');
        else read(l),read(r),read(k),write(tree.qus_nxt(l,r,k),'\n');
    }
    return 0;
}

template<typename T>
void read(T& x){
    x=0; bool f=0; char ch=getchar();
    while (ch<'0'||ch>'9'){
        if (ch=='-') f=1;
        ch=getchar();
    }
    while (ch>='0'&&ch<='9') x=x*10+ch-'0',ch=getchar();
    x=f?-x:x;
}
template<typename T>
void write(T x){
    if (x<0) putchar('-'),x=-x;
    if (x>9) write(x/10);
    putchar(x%10+'0');
}
template<typename T>
void write(T x,char c){
    write(x);
    putchar(c);
}
treap::treap(){
}
pair<int,int> treap::split(int p,int pos){ //分裂
    if (!p) return mp(0,0);
    if (tr[p].pos<=pos){
        auto temp=split(tr[p].r,pos);
        tr[p].r=temp.fi;
        upd(p);
        return mp(p,temp.se);
    }
    else{
        auto temp=split(tr[p].l,pos);
        tr[p].l=temp.se;
        upd(p);
        return mp(temp.fi,p);
    }
}
int treap::newnode(int pos){ //生成新节点
    ++cnt;
    tr[cnt].pos=pos; tr[cnt].siz=1;
    tr[cnt].pri=ran();
    return cnt;
}
int treap::merge(int u,int v){ //合并节点
    if (!u) return v;
    if (!v) return u;
    if (tr[u].pri>tr[v].pri){
        tr[u].r=merge(tr[u].r,v);
        upd(u);
        return u;
    }
    else {
        tr[v].l=merge(u,tr[v].l);
        upd(v);
        return v;
    }
}
void treap::upd(int p){ //更新子树大小
    tr[p].siz=tr[tr[p].l].siz+tr[tr[p].r].siz+1;
}
void treap::insert(int pos){ //插入节点
    auto temp1=split(root,pos);
    auto temp2=split(temp1.fi,pos-1);
    temp2.se=newnode(pos);
    root=merge(merge(temp2.fi,temp2.se),temp1.se);
}
void treap::erase(int pos){ //删除节点
    auto temp1=split(root,pos);
    auto temp2=split(temp1.fi,pos-1);
    temp2.se=0;
    root=merge(merge(temp2.fi,temp2.se),temp1.se);
}
int treap::qus(int l,int r){
    auto temp1=split(root,l-1);
    auto temp2=split(temp1.se,r);
    int ans=tr[temp2.fi].siz;
    root=merge(temp1.fi,merge(temp2.fi,temp2.se));
    return ans;
}
bst::bst(){
    cnt=1;
}
int bst::newnode(){ //新建节点
    ++cnt;
    return cnt;
}
int bst::add(int p,int l,int r,int k,int v){ //对应位置加入值
    if (!p) p=newnode();
    t[p].tr.insert(k);
    if (l==r) return p;
    int mid=l+r>>1;
    if (v<=mid) t[p].l=add(t[p].l,l,mid,k,v);
    else t[p].r=add(t[p].r,mid+1,r,k,v);
    return p;
}
int bst::del(int p,int l,int r,int k,int v){ //对应位置减少值
    if (!p) p=newnode();
    t[p].tr.erase(k);
    if (l==r) return p;
    int mid=l+r>>1;
    if (v<=mid) t[p].l=del(t[p].l,l,mid,k,v);
    else t[p].r=del(t[p].r,mid+1,r,k,v);
    return p;
}
int bst::qus_num(int p,int l,int r,int ql,int qr,int pl,int pr){ //询问区间数的个数
    if (!p) return 0;
    if (r<ql||l>qr) return 0;
    if (l>=ql&&r<=qr) return t[p].tr.qus(pl,pr);
    int mid=l+r>>1;
    return qus_num(t[p].l,l,mid,ql,qr,pl,pr)+qus_num(t[p].r,mid+1,r,ql,qr,pl,pr);
}
int bst::qus_rk(int p,int l,int r,int pl,int pr,int k){ //查询区间排名为某值的数
    if (l==r) return l;
    int mid=l+r>>1;
    int v=t[t[p].l].tr.qus(pl,pr);
    if (k<=v) return qus_rk(t[p].l,l,mid,pl,pr,k);
    else return qus_rk(t[p].r,mid+1,r,pl,pr,k-v);
}
int bst::qus_pre(int pl,int pr,int k){ //查询区间数的前驱
    int rk=qus_num(1,L,R,L,k-1,pl,pr);
    return rk?qus_rk(1,L,R,pl,pr,rk):-2147483647;
}
int bst::qus_nxt(int pl,int pr,int k){ //查询区间数的后驱
    int rk=qus_num(1,L,R,L,k,pl,pr);
    return (rk<pr-pl+1)?qus_rk(1,L,R,pl,pr,rk+1):2147483647;
}
```
